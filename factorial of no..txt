#include <stdio.h>

int main() {
    int n, i;
    // Use unsigned long long to handle potentially large factorial values
    unsigned long long factorial = 1;

    printf("Enter a positive integer: ");
    scanf("%d", &n);

    // Check if the number is a non-negative integer
    if (n < 0) {
        printf("Error! Factorial of a negative number doesn't exist.\\n");
    } else {
        // 0! is defined as 1, so the loop handles this correctly if n is 0
        for (i = 1; i <= n; ++i) {
            factorial *= i; // Multiply 'factorial' by 'i' in each iteration
        }
        printf("Factorial of %d = %llu\\n", n, factorial);
    }

    return 0;
}

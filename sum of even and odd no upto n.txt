#include <stdio.h>

void main()
{
    int i, n, odd_sum = 0, even_sum = 0;
    printf("Enter the value of n: ");
    scanf("%d", &n);

    for (i = 1; i <= n; i++)
    {
    
        if (i % 2 == 0)
        {
            even_sum = even_sum + i;
        }
    
        else
        {
            odd_sum = odd_sum + i;
        }
    }


    printf("Sum of all odd numbers from 1 to %d = %d\n", n, odd_sum);
    printf("Sum of all even numbers from 1 to %d = %d\n", n, even_sum);
}

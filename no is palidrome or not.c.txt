#include <stdio.h>

int main() {
    int n, original_n, reversed_n = 0, remainder;

    printf("Enter an integer: ");
    scanf("%d", &n);

    original_n = n; // Store the original number for comparison

    // Reverse the number
    while (n != 0) {
        remainder = n % 10;           // Extract the last digit
        reversed_n = reversed_n * 10 + remainder; // Add the digit to the reverse number
        n /= 10;                      // Remove the last digit
    }

    // Check if the original number is equal to the reversed number
    if (original_n == reversed_n) {
        printf("%d is a palindrome number.\n", original_n);
    } else {
        printf("%d is not a palindrome number.\n", original_n);
    }

    return 0;
}
